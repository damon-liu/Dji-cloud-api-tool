```cpp
void TopicManager::setAllTopicsEnabled(const QString& deviceSn, bool enabled) {
    if (!mDeviceTopics.contains(deviceSn))
        return;

    const QStringList topics = mDeviceTopics[deviceSn];
    if (topics.isEmpty())
        return;

    QStringList toAdd;
    QStringList toRemove;

    for (const auto& topic : topics) {
        bool currentlyEnabled = !mDisabledTopics.value(deviceSn).contains(topic);
        if (enabled && !currentlyEnabled) {
            mDisabledTopics[deviceSn].remove(topic);
            toAdd.append(topic);
        } else if (!enabled && currentlyEnabled) {
            mDisabledTopics[deviceSn].insert(topic);
            toRemove.append(topic);
        }
    }

    // 清理空的禁用集合
    if (mDisabledTopics.contains(deviceSn) && mDisabledTopics[deviceSn].isEmpty())
        mDisabledTopics.remove(deviceSn);

    // 批量发出变更信号（一次性重新订阅，而非逐条触发 N 次）
    if (!toAdd.isEmpty() || !toRemove.isEmpty())
        emit topicsChanged(toAdd, toRemove);
}
```

```cpp
void TopicListWidget::refreshList() {
    QString saved = selectedTopic();
    mTopicList->clear();
    mAddBtn->setEnabled(true);

    if (mCurrentSn.isEmpty()) {
        mTopicList->addItem(QString::fromUtf8("（请选择设备）"));
        mTopicList->item(0)->setFlags(Qt::NoItemFlags);
        mTopicList->item(0)->setForeground(QColor(180, 180, 180));
        mAddBtn->setEnabled(false);
        mToggleAllBtn->setEnabled(false);
        return;
    }

    if (mAllTopics.isEmpty()) {
        mTopicList->addItem(QString::fromUtf8("（无 Topic）"));
        mTopicList->item(0)->setFlags(Qt::NoItemFlags);
        mTopicList->item(0)->setForeground(QColor(180, 180, 180));
        mToggleAllBtn->setEnabled(false);
        return;
    }

    mToggleAllBtn->setEnabled(true);

    for (const auto& t : mAllTopics) {
        bool enabled = !mDisabledTopics.contains(t);
        QString prefix = enabled ? QString::fromUtf8("● ") : QString::fromUtf8("○ ");
        auto* item = new QListWidgetItem(prefix + t);
        item->setData(Qt::UserRole, t);
        if (!enabled)
            item->setForeground(QColor(180, 180, 180));
        mTopicList->addItem(item);
    }

    if (!saved.isEmpty()) {
        for (int i = 0; i < mTopicList->count(); ++i) {
            if (mTopicList->item(i)->data(Qt::UserRole).toString() == saved) {
                mTopicList->setCurrentRow(i);
                return;
            }
        }
    }
    if (mTopicList->count() > 0)
        mTopicList->setCurrentRow(0);
}
```

        mAutoPaused = false;
        if (!mPaused) {
            mRefreshTimer->start(mRefreshTimer->interval());
            refresh();
        }
    }
}
